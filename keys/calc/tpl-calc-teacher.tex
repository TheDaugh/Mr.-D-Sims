\subsection*{Targets}\begin{itemize}\item I can state the power rule in words.
\item I can differentiate a polynomial term by term.
\item I can find the slope at a point.\end{itemize}
\subsection*{Predictions}\begin{itemize}
\item The derivative of $x^2$ is $2x$. What do you expect for $x^3$? \A{$3x^2$.} Bring the 3 down and lower the power: $3x^2$.
\item What is the slope of the flat line $y=7$? \A{0.} A flat line does not rise: the derivative is 0.
\end{itemize}
\section*{Huddle script (answers in green)}
\subsection*{Definitions}\begin{itemize}
\item \textbf{Derivative:} The \A{slope} of the tangent line at a point.
\item \textbf{Power rule:} Bring the \A{power} down, then \A{subtract 1} from it.
\end{itemize}
\subsection*{Worked example}
Differentiate f(x) = 4x\textsuperscript{3}.\\
\begin{itemize}
\item Bring down the power: 3 $\cdot$ 4x
\item Lower the power by 1: 12x\A{2}
\item So f?(x) = \A{12x\textsuperscript{2}}
\end{itemize}
\subsection*{Label the slope}
\begin{enumerate}
\item The \A{point} where we measure the slope.
\item The \A{tangent line}; its slope is f?.
\end{enumerate}
\section*{Every level, with answers}
\subsection*{Differentiate}
\noindent\textbf{Level 1.} Differentiate.\\ \A{Answer: $4 x^{3}$} \quad Bring down 4 and lower the power: $4x^{3}$.\par\medskip
\noindent\textbf{Level 2.} Differentiate.\\ \A{Answer: $12 x$} \quad Multiply 6 by 2 to get 12, then lower the power: $12x$.\par\medskip
\noindent\textbf{Level 3.} Differentiate term by term.\\ \A{Answer: $10 x^{4} + 5$} \quad $10x^{4}+5+0$. The constant 1 disappears.\par\medskip
\noindent\textbf{Level 4.} Differentiate. Rewrite the root as a power if that helps.\\ \A{Answer: $\frac{6}{2 \sqrt{x}}$} \quad $6x^{1/2}$ becomes $\tfrac{6}{2}x^{-1/2}=\dfrac{6}{2\sqrt{x}}$.\par\medskip
\subsection*{Slope at a point}
\noindent\textbf{Level 1.} Let $f(x)=x^{2}$. Find the slope of the curve at $x=2$.\\ \A{Answer: 4} \quad $f'(x)=2x$, so $f'(2)=4$.\par\medskip
\noindent\textbf{Level 2.} Let $f(x)=4x^{2}$. Find the slope of the curve at $x=3$.\\ \A{Answer: 24} \quad $f'(x)=8x$, so $f'(3)=24$.\par\medskip
\noindent\textbf{Level 3.} Let $f(x)=2x^{4}$. Find the slope of the curve at $x=3$.\\ \A{Answer: 216} \quad $f'(x)=8x^{3}$, so $f'(3)=216$.\par\medskip
\noindent\textbf{Level 4.} Let $f(x)=4x^{4}$. Find the slope of the curve at $x=1$.\\ \A{Answer: 16} \quad $f'(x)=16x^{3}$, so $f'(1)=16$.\par\medskip
